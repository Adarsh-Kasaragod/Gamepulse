const API = "/api";
let allPlayers = [];
let allTeams = [];
let allMatches = [];

// Initialize page
document.addEventListener('DOMContentLoaded', function() {
    loadManageData();
    setupCalculators();
    setupFormValidation();
    setupRealTimeUpdates();
    startActivityStream();

// Load all data for dropdowns and stats
async function loadManageData() {
    try {
        showLoadingState();
        
        const [teamsRes, playersRes, matchesRes] = await Promise.all([
            fetch(`${API}/teams`).then(r => r.json()),
            fetch(`${API}/players`).then(r => r.json()),
            fetch(`${API}/matches?limit=10`).then(r => r.json())
        ]);

        // Store data globally
        allTeams = teamsRes.teams || [];
        allPlayers = playersRes.players || [];
        allMatches = matchesRes.matches || [];

        // Populate dropdowns
        populateTeamDropdown();
        populatePlayerDropdown();
        populateMatchDropdown();
        autoSelectLatestMatch();
        
        // Update statistics
        updateStatistics();
        
        // Load recent activity
        loadRecentActivity();
        
        hideLoadingState();
        showNotification('Database loaded successfully', 'success');
        
    } catch (error) {
        console.error('Error loading manage data:', error);
        showNotification('Failed to load data. Please refresh.', 'error');
    }
}

// Populate team dropdown
function populateTeamDropdown() {
    const teamSelect = document.getElementById('newPlayerTeam');
    if (!teamSelect) return;
    
    let options = '<option value="" disabled selected class="bg-dark-800">Select Team</option>';
    
    if (allTeams && allTeams.length > 0) {
        allTeams.forEach(team => {
            const playerCount = team.player_count || 0;
            const badge = playerCount > 0 ? 
                `<span class="text-green-400">(${playerCount} players)</span>` : 
                `<span class="text-gray-500">(No players)</span>`;
            options += `<option value="${team.id}" class="bg-dark-800">${team.name} ${badge}</option>`;
        });
    } else {
        options += '<option value="" disabled class="bg-dark-800">No teams available</option>';
    }
    
    teamSelect.innerHTML = options;
}

// Populate player dropdown
function populatePlayerDropdown() {
    const playerSelect = document.getElementById('perfPlayer');
    if (!playerSelect) return;
    
    let options = '<option value="" disabled selected class="bg-dark-800">Select Player</option>';
    
    if (allPlayers && allPlayers.length > 0) {
        // Sort players by name
        allPlayers.sort((a, b) => a.name.localeCompare(b.name));
        
        allPlayers.forEach(player => {
            const teamName = getTeamName(player.team_id);
            const roleColor = getRoleColor(player.role);
            options += `
                <option value="${player.id}" class="bg-dark-800">
                    <span class="${roleColor}">${player.name}</span> - ${player.role} (${teamName})
                </option>`;
        });
    } else {
        options += '<option value="" disabled class="bg-dark-800">No players available</option>';
    }
    
    playerSelect.innerHTML = options;
}

// Populate match dropdown
function populateMatchDropdown() {
    const matchSelect = document.getElementById('perfMatch');
    if (!matchSelect) return;
    
    let options = '<option value="" disabled selected class="bg-dark-800">Select Match</option>';
    
    if (allMatches && allMatches.length > 0) {
        // Sort matches by date (newest first)
        allMatches.sort((a, b) => new Date(b.date || 0) - new Date(a.date || 0));
        
        allMatches.forEach(match => {
            const date = match.date ? formatDate(match.date) : 'Unknown Date';
            const teams = match.team1_id && match.team2_id ? 
                `${getTeamName(match.team1_id)} vs ${getTeamName(match.team2_id)}` : 
                match.name;
            const formattedDate = `<span class="text-amber-400">${date}</span>`;
            options += `<option value="${match.id}" class="bg-dark-800">${teams} - ${formattedDate}</option>`;
        });
    } else {
        options += '<option value="" disabled class="bg-dark-800">No matches available</option>';
    }
    
    matchSelect.innerHTML = options;
}

// Get team name from team ID
function getTeamName(teamId) {
    const team = allTeams.find(t => t.id === teamId);
    return team ? team.name : 'Unknown Team';
}

// Get role color
function getRoleColor(role) {
    switch((role || '').toLowerCase()) {
        case 'batsman': return 'text-red-400';
        case 'bowler': return 'text-blue-400';
        case 'all-rounder': return 'text-green-400';
        case 'wicket-keeper': return 'text-purple-400';
        default: return 'text-gray-400';
    }
}

// Update statistics display
function updateStatistics() {
    const totalPlayers = allPlayers.length;
    const totalTeams = allTeams.length;
    const totalMatches = allMatches.length;
    
    // Calculate total performances (this would need a separate API endpoint)
    // For now, estimate based on players * matches
    const estimatedPerformances = Math.round(totalPlayers * totalMatches * 0.5);
    
    document.getElementById('totalPlayersStat').textContent = totalPlayers;
    document.getElementById('totalTeamsStat').textContent = totalTeams;
    document.getElementById('totalMatchesStat').textContent = totalMatches;
    document.getElementById('totalPerfStat').textContent = estimatedPerformances;
    
    document.getElementById('dataStats').textContent = 
        `${totalPlayers} players • ${totalTeams} teams • ${totalMatches} matches`;
}

// Setup real-time calculators
function setupCalculators() {
    // Batting calculators
    const battingInputs = ['perfRuns', 'perfBalls'];
    battingInputs.forEach(id => {
        const input = document.getElementById(id);
        if (input) {
            input.addEventListener('input', updateBattingStats);
            input.addEventListener('input', animateCalculation);
        }
    });
    
    // Bowling calculators
    const bowlingInputs = ['perfOvers', 'perfWickets', 'perfConceded'];
    bowlingInputs.forEach(id => {
        const input = document.getElementById(id);
        if (input) {
            input.addEventListener('input', updateBowlingStats);
            input.addEventListener('input', animateCalculation);
        }
    });
}

// Animate calculation updates
function animateCalculation() {
    const target = event.target.id.includes('Runs') || event.target.id.includes('Balls') ? 
                   'strikeRateCalc' : 'economyCalc';
    const element = document.getElementById(target);
    if (element) {
        element.classList.add('animate-pulse');
        setTimeout(() => element.classList.remove('animate-pulse'), 300);
    }
}

// Update batting statistics in real-time
function updateBattingStats() {
    const runs = parseInt(document.getElementById('perfRuns').value) || 0;
    const balls = parseInt(document.getElementById('perfBalls').value) || 0;
    
    // Calculate strike rate
    const strikeRate = balls > 0 ? ((runs / balls) * 100).toFixed(2) : '0.00';
    const strikeRateElement = document.getElementById('strikeRateCalc');
    if (strikeRateElement) {
        strikeRateElement.textContent = strikeRate;
        strikeRateElement.classList.add('text-green-400');
        if (parseFloat(strikeRate) > 150) {
            strikeRateElement.classList.add('neon-glow');
        } else {
            strikeRateElement.classList.remove('neon-glow');
        }
    }
    
    // Calculate batting average (assuming 1 dismissal per innings)
    const battingAvg = runs > 0 ? runs.toFixed(2) : '0.00';
    const battingAvgElement = document.getElementById('battingAvgCalc');
    if (battingAvgElement) {
        battingAvgElement.textContent = battingAvg;
    }
}

// Update bowling statistics in real-time
function updateBowlingStats() {
    const overs = parseFloat(document.getElementById('perfOvers').value) || 0;
    const wickets = parseInt(document.getElementById('perfWickets').value) || 0;
    const runs = parseInt(document.getElementById('perfConceded').value) || 0;
    
    // Calculate economy
    const economy = overs > 0 ? (runs / overs).toFixed(2) : '0.00';
    const economyElement = document.getElementById('economyCalc');
    if (economyElement) {
        economyElement.textContent = economy;
        if (parseFloat(economy) < 6.0) {
            economyElement.classList.add('text-green-400', 'neon-glow');
            economyElement.classList.remove('text-amber-400');
        } else {
            economyElement.classList.remove('neon-glow');
            economyElement.classList.add('text-amber-400');
        }
    }
    
    // Calculate bowling average
    const bowlingAvg = wickets > 0 ? (runs / wickets).toFixed(2) : '0.00';
    const bowlingAvgElement = document.getElementById('bowlingAvgCalc');
    if (bowlingAvgElement) {
        bowlingAvgElement.textContent = bowlingAvg;
        if (parseFloat(bowlingAvg) < 25.0) {
            bowlingAvgElement.classList.add('text-green-400', 'neon-glow');
            bowlingAvgElement.classList.remove('text-purple-400');
        } else {
            bowlingAvgElement.classList.remove('neon-glow');
            bowlingAvgElement.classList.add('text-purple-400');
        }
    }
}

// Add activity item
function addActivityItem(user, message, type = 'info') {
    const activityContainer = document.getElementById('recentActivity');
    if (!activityContainer) return;
    
    const colors = {
        success: 'text-green-400',
        error: 'text-red-400',
        info: 'text-blue-400',
        warning: 'text-amber-400'
    };
    
    const icons = {
        success: 'fa-check-circle',
        error: 'fa-exclamation-circle',
        info: 'fa-info-circle',
        warning: 'fa-exclamation-triangle'
    };
    
    const now = new Date();
    const timeString = now.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
    
    const activityItem = document.createElement('div');
    activityItem.className = 'flex items-center gap-4 p-4 glass-panel rounded-xl border border-white/5 hover:border-white/10 transition-all duration-300 animate-fade-in';
    activityItem.innerHTML = `
        <div class="w-10 h-10 rounded-full bg-gradient-to-br from-blue-500/20 to-indigo-600/20 flex items-center justify-center flex-shrink-0">
            <i class="fas ${icons[type] || icons.info} ${colors[type] || colors.info}"></i>
        </div>
        <div class="flex-1 min-w-0">
            <div class="flex items-center justify-between mb-1">
                <span class="font-medium text-white">${user}</span>
                <span class="text-xs text-gray-500">${timeString}</span>
            </div>
            <p class="text-sm text-gray-400 truncate">${message}</p>
        </div>
    `;
    
    // Add to top of container
    if (activityContainer.firstChild) {
        activityContainer.insertBefore(activityItem, activityContainer.firstChild);
    } else {
        activityContainer.appendChild(activityItem);
    }
    
    // Limit to 5 items
    while (activityContainer.children.length > 5) {
        activityContainer.removeChild(activityContainer.lastChild);
    }
}

// Add new player to database
async function addNewPlayer() {
    const nameInput = document.getElementById('newPlayerName');
    const roleSelect = document.getElementById('newPlayerRole');
    const teamSelect = document.getElementById('newPlayerTeam');
    const specializationInput = document.getElementById('newPlayerSpecialization');
    const button = document.getElementById('addPlayerBtn');
    
    if (!nameInput || !roleSelect || !teamSelect || !button) return;
    
    const name = nameInput.value.trim();
    const role = roleSelect.value;
    const team_id = teamSelect.value;
    const specialization = specializationInput.value.trim();
    
    // Validation
    if (!name) {
        showNotification('Please enter player name', 'error');
        return;
    }
    
    if (!team_id) {
        showNotification('Please select a team', 'error');
        return;
    }
    
    if (!role) {
        showNotification('Please select a role', 'error');
        return;
    }
    
    // Disable button and show loading
    const originalHTML = button.innerHTML;
    button.disabled = true;
    button.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Adding Player...';
    button.classList.add('animate-pulse');
    
    try {
        const playerData = { 
            name, 
            role, 
            team_id: parseInt(team_id)
        };
        
        // Add specialization if provided
        if (specialization) {
            playerData.specialization = specialization;
        }
        
        const res = await fetch(`${API}/players`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(playerData)
        });
        
        const data = await res.json();
        
        if (data.success) {
            // Add activity
            addActivityItem('You', `Added player "${name}"`, 'success');
            
            showSuccessModal(
                'Player Added Successfully!',
                `Player "${name}" has been registered in the database.`,
                'player'
            );
            
            // Reset form
            nameInput.value = '';
            specializationInput.value = '';
            teamSelect.value = '';
            roleSelect.value = '';
            
            // Refresh data
            setTimeout(() => {
                loadManageData();
                showNotification('Player database updated', 'success');
            }, 1000);
        } else {
            throw new Error(data.error || 'Failed to add player');
        }
    } catch (error) {
        console.error('Add player error:', error);
        addActivityItem('System', `Failed to add player: ${error.message}`, 'error');
        showNotification(`Error: ${error.message}`, 'error');
    } finally {
        // Re-enable button
        button.disabled = false;
        button.innerHTML = originalHTML;
        button.classList.remove('animate-pulse');
    }
}

// Add performance data to database
async function addPerformance() {
    const playerSelect = document.getElementById('perfPlayer');
    const matchSelect = document.getElementById('perfMatch');
    const button = document.getElementById('addPerfBtn');
    
    if (!playerSelect || !matchSelect || !button) return;
    
    const player_id = playerSelect.value;
    const match_id = matchSelect.value;
    
    // Validation
    if (!player_id || !match_id) {
        showNotification('Please select both player and match', 'error');
        return;
    }
    
    const battingData = {
        match_id: parseInt(match_id),
        player_id: parseInt(player_id),
        runs: parseInt(document.getElementById('perfRuns').value) || 0,
        balls_faced: parseInt(document.getElementById('perfBalls').value) || 0
    };
    
    const bowlingData = {
        match_id: parseInt(match_id),
        player_id: parseInt(player_id),
        overs: parseFloat(document.getElementById('perfOvers').value) || 0,
        wickets: parseInt(document.getElementById('perfWickets').value) || 0,
        runs_conceded: parseInt(document.getElementById('perfConceded').value) || 0
    };
    
    const hasBatting = battingData.runs > 0 || battingData.balls_faced > 0;
    const hasBowling = bowlingData.overs > 0 || bowlingData.wickets > 0 || bowlingData.runs_conceded > 0;
    
    if (!hasBatting && !hasBowling) {
        showNotification('Please enter at least some performance data', 'error');
        return;
    }
    
    // Disable button and show loading
    const originalHTML = button.innerHTML;
    button.disabled = true;
    button.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Saving...';
    button.classList.add('animate-pulse');
    
    try {
        const requests = [];
        let savedData = { batting: false, bowling: false };
        
        if (hasBatting) {
            const battingRes = await fetch(`${API}/performances/batting`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(battingData)
            });
            
            const battingResult = await battingRes.json();
            savedData.batting = battingResult.success;
            
            if (!battingResult.success) {
                throw new Error('Failed to save batting data');
            }
        }
        
        if (hasBowling) {
            const bowlingRes = await fetch(`${API}/performances/bowling`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(bowlingData)
            });
            
            const bowlingResult = await bowlingRes.json();
            savedData.bowling = bowlingResult.success;
            
            if (!bowlingResult.success) {
                throw new Error('Failed to save bowling data');
            }
        }
        
        const playerName = playerSelect.options[playerSelect.selectedIndex].text.split(' - ')[0];
        const successMessage = hasBatting && hasBowling ? 
            'Batting and bowling data saved' :
            hasBatting ? 'Batting data saved' : 'Bowling data saved';
        
        // Add activity
        addActivityItem('You', `${successMessage} for ${playerName}`, 'success');
            
        showSuccessModal(
            'Performance Recorded!',
            `${successMessage} for ${playerName}`,
            'performance'
        );
        
        // Reset form
        resetPerformanceForm();
        
        // Refresh data
        setTimeout(() => {
            loadRecentActivity();
            showNotification('Performance database updated', 'success');
        }, 1000);
        
    } catch (error) {
        console.error('Add performance error:', error);
        addActivityItem('System', `Failed to save performance: ${error.message}`, 'error');
        showNotification(`Error: ${error.message}`, 'error');
    } finally {
        // Re-enable button
        button.disabled = false;
        button.innerHTML = originalHTML;
        button.classList.remove('animate-pulse');
    }
}

// Utility functions
function formatDate(dateString) {
    if (!dateString) return 'Unknown Date';
    try {
        const date = new Date(dateString);
        return date.toLocaleDateString('en-US', { 
            month: 'short', 
            day: 'numeric',
            year: 'numeric'
        });
    } catch (error) {
        return dateString;
    }
}

// Create confetti effect
function createConfetti() {
    const colors = ['#3b82f6', '#8b5cf6', '#10b981', '#f59e0b', '#ef4444'];
    
    for (let i = 0; i < 50; i++) {
        const confetti = document.createElement('div');
        confetti.style.position = 'fixed';
        confetti.style.width = '10px';
        confetti.style.height = '10px';
        confetti.style.background = colors[Math.floor(Math.random() * colors.length)];
        confetti.style.borderRadius = '50%';
        confetti.style.left = `${Math.random() * 100}vw`;
        confetti.style.top = '-10px';
        confetti.style.zIndex = '9999';
        confetti.style.opacity = '0.8';
        
        document.body.appendChild(confetti);
        
        // Animate
        const animation = confetti.animate([
            { transform: 'translateY(0) rotate(0deg)', opacity: 1 },
            { transform: `translateY(${window.innerHeight}px) rotate(${Math.random() * 360}deg)`, opacity: 0 }
        ], {
            duration: 2000 + Math.random() * 2000,
            easing: 'cubic-bezier(0.215, 0.61, 0.355, 1)'
        });
        
        animation.onfinish = () => confetti.remove();
    }
}

    };
    
    const icons = {
        success: 'fa-check-circle',
        error: 'fa-exclamation-circle',

    };
    
    const toast = document.createElement('div');
    toast.className = `toast-notification fixed top-6 right-6 px-6 py-4 rounded-xl shadow-2xl z-50 animate-slide-down backdrop-blur-lg ${colors[type] || colors.info} border`;
